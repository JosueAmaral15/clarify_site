## 🏗️ Architecture & Design

### Component Hierarchy
```
Application (app.py)
├── ShortcutManager (singleton, static class)
│   ├── SHORTCUTS dict (45+ default shortcuts)
│   ├── OS_SHORTCUTS dict (platform-specific conflicts)
│   └── _custom_shortcuts dict (user customizations)
├── ShortcutsDialog (QDialog)
│   ├── QTableWidget (4 columns)
│   ├── Search/Filter (QLineEdit)
│   ├── Edit Dialog (QKeySequenceEdit)
│   └── Buttons (Reset Selected, Reset All, Save, Cancel)
├── SettingsManager
│   ├── get_custom_shortcuts() -> Dict[str, str]
│   └── set_custom_shortcuts(dict)
└── Navigation System
    ├── keyPressEvent() - global handler
    ├── _get_all_docks() - enumerate docks
    ├── _get_visible_docks() - filter visible
    ├── _get_dock_by_menu_order() - ordered list [1-9]
    ├── _navigate_next_dock() - Ctrl+Tab
    ├── _navigate_prev_dock() - Ctrl+Shift+Tab
    ├── _navigate_next_dock_dynamic() - Ctrl+PageDown
    ├── _navigate_prev_dock_dynamic() - Ctrl+PageUp
    ├── _jump_to_dock(index) - Ctrl+1-9
    └── _focus_dock(dock) - activate and focus
```

### Design Patterns Applied

1. **Command Pattern**
   - Each shortcut customization is a reversible command
   - Pending changes stored in `_pending_changes` dict
   - Can be reset individually or in batch

2. **Observer Pattern**
   - ShortcutsDialog emits `shortcuts_changed` signal
   - App listens via `_on_shortcuts_changed()` slot
   - Updates applied to all menu items automatically

3. **Strategy Pattern**
   - Different navigation strategies (menu order vs dynamic)
   - Pluggable conflict detection (OS-specific)

4. **Singleton**
   - ShortcutManager as static class (single source of truth)
   - No instances needed, all methods classmethod

5. **Information Expert (GRASP)**
   - ShortcutManager knows all shortcuts, categories, conflicts
   - Encapsulates knowledge about keyboard mappings

6. **Low Coupling**
   - ShortcutsDialog doesn't know about App
   - Communication via signals/slots
   - ShortcutManager independent of GUI

7. **High Cohesion**
   - Each component has single, well-defined responsibility
   - ShortcutManager: shortcuts only
   - ShortcutsDialog: UI only
   - Navigation methods: navigation only

```python
@classmethod
def get_default_shortcut(cls, action_name: str) -> Tuple[str, str]:
    """Get default (non-customized) shortcut."""
    return cls.SHORTCUTS.get(action_name, ("", ""))

@classmethod
def set_custom_shortcut(cls, action_name: str, key_sequence: str) -> None:
    """Set custom shortcut for an action."""
    conflicts = cls.get_conflicts(key_sequence)
    cls._custom_shortcuts[action_name] = key_sequence

@classmethod
def reset_custom_shortcut(cls, action_name: str) -> None:
    """Reset to default shortcut."""
    if action_name in cls._custom_shortcuts:
        del cls._custom_shortcuts[action_name]

@classmethod
def load_custom_shortcuts(cls, custom_dict: Dict[str, str]) -> None:
    """Load custom shortcuts from settings."""
    cls._custom_shortcuts = custom_dict.copy()

@classmethod
def get_conflicts(cls, key_sequence: str) -> List[str]:
    """Check for conflicting shortcuts."""
    conflicts = []
    for action_name, (shortcut, _) in cls.get_all_shortcuts().items():
        if shortcut.lower() == key_sequence.lower():
            conflicts.append(action_name)
    return conflicts

@classmethod
def is_os_shortcut(cls, key_sequence: str) -> bool:
    """Check if shortcut conflicts with OS."""
    system = platform.system()
    os_shortcuts = cls.OS_SHORTCUTS.get(system, [])
    return key_sequence in os_shortcuts

@classmethod
def get_category(cls, action_name: str) -> str:
    """Get category for an action."""
    # Logic based on SHORTCUTS position
    return "File" | "Edit" | "Tools" | "Navigation" | "View" | "Processing" | "Help"
```

**1. keyPressEvent() - Global Navigation Handler**
```python
def keyPressEvent(self, event):
    """Handle global keyboard shortcuts (v2.9.33)."""
    from PySide6.QtCore import Qt
    
    key = event.key()
    modifiers = event.modifiers()
    
    # Ctrl+Tab - Next dock (menu order)
    if key == Qt.Key_Tab and modifiers == Qt.ControlModifier:
        self._navigate_next_dock()
        event.accept()
        return
    
    # Ctrl+Shift+Tab - Previous dock (menu order)
    elif key == Qt.Key_Tab and modifiers == (Qt.ControlModifier | Qt.ShiftModifier):
        self._navigate_prev_dock()
        event.accept()
        return
    
    # Ctrl+PageDown - Next dock (dynamic/creation order)
    elif key == Qt.Key_PageDown and modifiers == Qt.ControlModifier:
        self._navigate_next_dock_dynamic()
        event.accept()
        return
    
    # Ctrl+PageUp - Previous dock (dynamic/creation order)
    elif key == Qt.Key_PageUp and modifiers == Qt.ControlModifier:
        self._navigate_prev_dock_dynamic()
        event.accept()
        return
    
    # Ctrl+1-9 - Jump to specific dock
    elif modifiers == Qt.ControlModifier:
        if Qt.Key_1 <= key <= Qt.Key_9:
            dock_index = key - Qt.Key_1  # 0-based
            self._jump_to_dock(dock_index)
            event.accept()
            return
    
    # F11 - Fullscreen
    elif key == Qt.Key_F11:
        if self.isFullScreen():
            self.showNormal()
        else:
            self.showFullScreen()
        event.accept()
        return
    
    # Pass to parent for default handling
    super().keyPressEvent(event)
```

```python
def _get_all_docks(self):
    """Get list of all QDockWidget instances."""
    docks = []
    for attr_name in dir(self):
        if attr_name.startswith('dock_'):
            dock = getattr(self, attr_name, None)
            if dock and isinstance(dock, QDockWidget):
                docks.append(dock)
    return docks

def _get_visible_docks(self):
    """Get list of visible QDockWidget instances."""
    return [dock for dock in self._get_all_docks() if dock.isVisible()]

def _get_dock_by_menu_order(self):
    """Get docks in menu order (Tools menu 1-9)."""
    ordered_names = [
        'dock_json_search',      # 1
        'dock_rewriter',         # 2
        'dock_set_operations',   # 3
        'dock_ai_processor',     # 4
        'dock_case_converter',   # 5
        'dock_text_extractor',   # 6
        'dock_text_to_json',     # 7
        'dock_placeholder_replacer',  # 8
        'dock_translate',        # 9
    ]
    
    docks = []
    for name in ordered_names:
        dock = getattr(self, name, None)
        if dock and isinstance(dock, QDockWidget):
            docks.append(dock)
    return docks

def _navigate_next_dock(self):
    """Navigate to next dock in menu order."""
    # Get visible docks in menu order
    visible_docks = self._get_visible_docks()
    if not visible_docks:
        return
    
    # Find current focused dock
    focused_dock = None
    for dock in visible_docks:
        if dock.widget() and dock.widget().hasFocus():
            focused_dock = dock
            break
    
    # Get all docks in menu order
    ordered_docks = self._get_dock_by_menu_order()
    visible_ordered = [d for d in ordered_docks if d in visible_docks]
    
    if not visible_ordered:
        return
    
    # Find next dock (circular)
    if focused_dock and focused_dock in visible_ordered:
        current_idx = visible_ordered.index(focused_dock)
        next_idx = (current_idx + 1) % len(visible_ordered)
        next_dock = visible_ordered[next_idx]
    else:
        next_dock = visible_ordered[0]
    
    self._focus_dock(next_dock)

def _navigate_prev_dock(self):
    """Navigate to previous dock in menu order."""
    # Similar to _navigate_next_dock but reverse direction

def _navigate_next_dock_dynamic(self):
    """Navigate to next dock in creation order."""
    visible_docks = self._get_visible_docks()
    if not visible_docks:
        return
    
    # Find current focused dock
    focused_dock = None
    for dock in visible_docks:
        if dock.widget() and dock.widget().hasFocus():
            focused_dock = dock
            break
    
    # Find next dock (circular in visible_docks list)
    if focused_dock and focused_dock in visible_docks:
        current_idx = visible_docks.index(focused_dock)
        next_idx = (current_idx + 1) % len(visible_docks)
        next_dock = visible_docks[next_idx]
    else:
        next_dock = visible_docks[0]
    
    self._focus_dock(next_dock)

def _navigate_prev_dock_dynamic(self):
    """Navigate to previous dock in creation order."""
    # Similar to _navigate_next_dock_dynamic but reverse

def _jump_to_dock(self, dock_index: int):
    """Jump to specific dock by index (0-based)."""
    ordered_docks = self._get_dock_by_menu_order()
    
    if 0 <= dock_index < len(ordered_docks):
        dock = ordered_docks[dock_index]
        
        # If dock is not visible, show it (Q10 answer)
        if not dock.isVisible():
            dock.show()
        
        self._focus_dock(dock)
    # If dock doesn't exist, ignore silently (Q3 answer)

def _focus_dock(self, dock: QDockWidget):
    """Focus on a dock widget."""
    if not dock or not dock.isVisible():
        return
    
    # Raise the dock
    dock.raise_()
    dock.activateWindow()
    
    # Focus the widget inside
    if dock.widget():
        dock.widget().setFocus()
    
    # Update status bar
    self.statusBar().showMessage(f"Focused: {dock.windowTitle()}", 2000)
```
